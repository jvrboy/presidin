    77→# Deriv-sourced symbols scan on their own interval (multiplied) since the
    78→# unified analysis engine is heavier per-symbol than the alert monitor and
    79→# some of the 16 symbols hit the live Deriv WebSocket API.
    80→_SYMBOL_SCAN_INTERVAL_SECONDS = 900
    81→
    82→
    83→async def alert_monitor() -> None:
    84→    while True:
    85→        await asyncio.sleep(settings.alert_poll_seconds)
    86→        if engine_instance is None:
    87→            continue
    88→        db = SessionLocal()
    89→        try:
    90→            for rule in db.scalars(select(AlertRule).where(AlertRule.enabled == True)).all():  # noqa: E712
    91→                try:
    92→                    evaluate_rule(db, rule, engine_instance.provider)
    93→                except Exception:
    94→                    db.rollback()
    95→        finally:
    96→            db.close()
    97→
    98→
    99→async def symbol_scanner() -> None:
   100→    """Background scheduler that periodically runs the unified per-symbol
   101→    analysis engine across all 16 Deriv-sourced symbols and persists each
   102→    result as an `AnalysisSnapshot` row -- the persistent-storage side of
   103→    "analyze each symbol and save all analysis to persistent storage"."""
   104→    while True:
   105→        await asyncio.sleep(_SYMBOL_SCAN_INTERVAL_SECONDS)
   106→        if engine_instance is None:
   107→            continue
   108→        db = SessionLocal()
   109→        try:
   110→            for symbol in settings.deriv_symbols:
   111→                try:
   112→                    rows = engine_instance.provider.get_ohlc(symbol, 300)
   113→                    result = analyze_symbol(rows, symbol=symbol, timeframe="1h", lookback=20)
   114→                    db.add(AnalysisSnapshot(symbol=symbol, timeframe="1h", bias=result["overall_bias"], confidence=result["overall_confidence"], commentary=result["commentary"], result_json=json.dumps(result, default=str)))
   115→                    db.commit()
   116→                except Exception:
   117→                    db.rollback()
   118→        finally:
   119→            db.close()

   122→@asynccontextmanager
   123→async def lifespan(_app):
   124→    global engine_instance, alert_monitor_task, symbol_scan_task
   125→    init_db()
   126→    db = next(get_db())
   127→    engine_instance = TradingEngine(db)
   128→    alert_monitor_task = asyncio.create_task(alert_monitor())
   129→    symbol_scan_task = asyncio.create_task(symbol_scanner())
   130→    yield
   131→    if alert_monitor_task:
   132→        alert_monitor_task.cancel()
   133→    if symbol_scan_task:
   134→        symbol_scan_task.cancel()
   135→    db.close()
   136→    engine_instance = None
